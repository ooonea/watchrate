cmake_minimum_required(VERSION 3.20)
project(watchrate VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql)
add_executable(watchrate src/main.cpp)
qt_add_resources(watchrate "icon" PREFIX "/" FILES data/watchrate.png)
target_link_libraries(watchrate PRIVATE Qt6::Widgets Qt6::Sql)
target_compile_options(watchrate PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS watchrate RUNTIME DESTINATION bin)
install(FILES data/dev.ooonea.Watchrate.desktop DESTINATION share/applications)
install(FILES data/watchrate.svg DESTINATION share/icons/hicolor/scalable/apps RENAME dev.ooonea.Watchrate.svg)
install(FILES data/watchrate.png DESTINATION share/icons/hicolor/512x512/apps RENAME dev.ooonea.Watchrate.png)
include(CTest)
add_test(NAME measurement COMMAND watchrate --self-test)
set_tests_properties(measurement PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
